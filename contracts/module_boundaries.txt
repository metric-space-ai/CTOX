# CTOX module boundaries — the single source for the import guard.
#
# Each line is `<from> -> <to> = <max>`, counting occurrences of `crate::<to>`
# anywhere under `src/core/<from>`. The guard in
# src/core/module_boundary_tests.rs reads this file and fails when a count
# exceeds its budget.
#
# `= 0` is an enforced boundary: that edge does not exist and must not appear.
# Any other number is a ratchet — the edge exists today, the number is what it
# costs right now, and it may only go down. Lowering a ratchet means editing
# this file in the same commit, which is the point: the number is visible in
# review instead of drifting quietly.
#
# Direction matters. `mission -> service` means code under src/core/mission
# reaching into src/core/service.

# --- Enforced: these edges do not exist ---------------------------------
# The capability, communication and context cores are below Business OS and
# below the service loop. They are driven, they do not reach upward.
capabilities -> business_os = 0
capabilities -> mission = 0
communication -> business_os = 0
communication -> mission = 0
communication -> service = 0
context -> business_os = 0
context -> service = 0
mission -> capabilities = 0
# The command lifecycle contract and transition validator live below Mission
# and Business OS, so Mission no longer reaches upward for lifecycle decisions.
mission -> business_os = 0

# --- Ratchets: existing edges, allowed to shrink only -------------------
# Mission still reaches into service for harness flow/mining and governance;
# the core state machine and transition guard now live below both modules.
mission -> service = 6
# mission drives communication adapters directly.
mission -> communication = 10
# Business OS still uses service-side harness flow and Business OS bridges.
business_os -> service = 12
# One capability reaches the service layer.
capabilities -> service = 1
# Context reads one mission type.
context -> mission = 1
