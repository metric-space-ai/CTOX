package gemini

import (
    "context"
    "encoding/json"
    "os"
    "testing"

    "github.com/router-for-me/CLIProxyAPI/v7/internal/signature"
)

type ctoxGeminiPassthroughFixture struct { Name string `json:"name"`; Operation string `json:"operation"`; Model string `json:"model"`; Stream bool `json:"stream"`; Input json.RawMessage `json:"input"`; RawSSE string `json:"raw_sse"`; RawText string `json:"raw_text"` }
type ctoxGeminiPassthroughResult struct { Name string `json:"name"`; Output any `json:"output"` }

func TestCtoxGeminiPassthroughDifferential(t *testing.T) {
    raw, err := os.ReadFile(os.Getenv("CTOX_DIFF_FIXTURES")); if err != nil { t.Fatal(err) }
    var fixtures []ctoxGeminiPassthroughFixture
    if err = json.Unmarshal(raw, &fixtures); err != nil { t.Fatal(err) }
    results := make([]ctoxGeminiPassthroughResult, 0, len(fixtures))
    for _, fixture := range fixtures {
        var output any
        switch fixture.Operation {
        case "gemini_passthrough_request":
            converted := ConvertGeminiRequestToGemini(fixture.Model, fixture.Input, fixture.Stream)
            if err = json.Unmarshal(converted, &output); err != nil { t.Fatal(err) }
        case "gemini_signature_sanitize":
            converted := signature.SanitizeGeminiRequestThoughtSignatures(fixture.Input, "contents")
            if err = json.Unmarshal(converted, &output); err != nil { t.Fatal(err) }
        case "gemini_passthrough_stream":
            converted := PassthroughGeminiResponseStream(context.Background(), fixture.Model, nil, nil, []byte(fixture.RawSSE), nil)
            values := make([]string, 0, len(converted)); for _, item := range converted { values = append(values, string(item)) }; output = values
        case "gemini_passthrough_non_stream":
            output = string(PassthroughGeminiResponseNonStream(context.Background(), fixture.Model, nil, nil, []byte(fixture.RawText), nil))
        case "gemini_passthrough_count":
            var value struct { Count int64 `json:"count"` }; if err = json.Unmarshal(fixture.Input, &value); err != nil { t.Fatal(err) }
            if err = json.Unmarshal(GeminiTokenCount(context.Background(), value.Count), &output); err != nil { t.Fatal(err) }
        default: t.Fatalf("unknown operation %s", fixture.Operation)
        }
        results = append(results, ctoxGeminiPassthroughResult{Name:fixture.Name, Output:output})
    }
    encoded, err := json.MarshalIndent(results, "", "  "); if err != nil { t.Fatal(err) }
    if err = os.WriteFile(os.Getenv("CTOX_DIFF_OUTPUT"), encoded, 0o600); err != nil { t.Fatal(err) }
}
