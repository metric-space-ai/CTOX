package chat_completions

import (
    "encoding/json"
    "os"
    "testing"
)

type ctoxClaudeChatRequestFixture struct {
    Name string `json:"name"`
    Operation string `json:"operation"`
    Model string `json:"model"`
    Stream bool `json:"stream"`
    Input json.RawMessage `json:"input"`
}

type ctoxClaudeChatRequestResult struct {
    Name string `json:"name"`
    Output any `json:"output"`
}

func TestCtoxClaudeChatRequestDifferential(t *testing.T) {
    raw, err := os.ReadFile(os.Getenv("CTOX_DIFF_FIXTURES"))
    if err != nil { t.Fatal(err) }
    var fixtures []ctoxClaudeChatRequestFixture
    if err = json.Unmarshal(raw, &fixtures); err != nil { t.Fatal(err) }
    results := make([]ctoxClaudeChatRequestResult, 0, len(fixtures))
    for _, fixture := range fixtures {
        if fixture.Operation != "chat_request_to_claude" {
            t.Fatalf("unknown operation %q", fixture.Operation)
        }
        output := ConvertOpenAIRequestToClaude(fixture.Model, fixture.Input, fixture.Stream)
        var value map[string]any
        if err = json.Unmarshal(output, &value); err != nil { t.Fatal(err) }
        delete(value, "metadata")
        results = append(results, ctoxClaudeChatRequestResult{Name: fixture.Name, Output: value})
    }
    encoded, err := json.MarshalIndent(results, "", "  ")
    if err != nil { t.Fatal(err) }
    if err = os.WriteFile(os.Getenv("CTOX_DIFF_OUTPUT"), encoded, 0o600); err != nil { t.Fatal(err) }
}
