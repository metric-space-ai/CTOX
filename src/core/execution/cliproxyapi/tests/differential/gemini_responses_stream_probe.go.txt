package responses

import (
    "bytes"
    "context"
    "encoding/json"
    "os"
    "strings"
    "testing"
)

type ctoxGeminiStreamFixture struct {
    Name string `json:"name"`
    Operation string `json:"operation"`
    Model string `json:"model"`
    OriginalRequest json.RawMessage `json:"original_request"`
    RawSSEChunks []string `json:"raw_sse_chunks"`
}

type ctoxGeminiStreamResult struct { Name string `json:"name"`; Output any `json:"output"` }

func TestCtoxGeminiResponsesStreamDifferential(t *testing.T) {
    raw, err := os.ReadFile(os.Getenv("CTOX_DIFF_FIXTURES")); if err != nil { t.Fatal(err) }
    var fixtures []ctoxGeminiStreamFixture; if err = json.Unmarshal(raw, &fixtures); err != nil { t.Fatal(err) }
    results := make([]ctoxGeminiStreamResult, 0, len(fixtures))
    for _, fixture := range fixtures {
        if fixture.Operation != "gemini_to_responses_stream" { t.Fatalf("unknown operation %q", fixture.Operation) }
        var state any
        events := make([]any, 0)
        for _, chunk := range fixture.RawSSEChunks {
            for _, rawEvent := range ConvertGeminiResponseToOpenAIResponses(
                context.Background(), fixture.Model, fixture.OriginalRequest, nil, []byte(chunk), &state,
            ) {
                eventName := ""
                var data []byte
                for _, line := range bytes.Split(rawEvent, []byte("\n")) {
                    if bytes.HasPrefix(line, []byte("event:")) { eventName = string(bytes.TrimSpace(line[6:])) }
                    if bytes.HasPrefix(line, []byte("data:")) { data = bytes.TrimSpace(line[5:]) }
                }
                var payload any; if err = json.Unmarshal(data, &payload); err != nil { t.Fatal(err) }
                events = append(events, map[string]any{"event":eventName,"data":payload})
            }
        }
        normalized := any(events); normalizeCtoxGeminiStream(normalized)
        results = append(results, ctoxGeminiStreamResult{Name:fixture.Name, Output:normalized})
    }
    encoded, err := json.MarshalIndent(results, "", "  "); if err != nil { t.Fatal(err) }
    if err = os.WriteFile(os.Getenv("CTOX_DIFF_OUTPUT"), encoded, 0o600); err != nil { t.Fatal(err) }
}

func normalizeCtoxGeminiStream(value any) {
    switch typed := value.(type) {
    case map[string]any:
        delete(typed, "created_at")
        if typed["type"] == "function_call" {
            typed["id"] = "fc_<dynamic>"
            typed["call_id"] = "call_<dynamic>"
        }
        if itemID, ok := typed["item_id"].(string); ok && strings.HasPrefix(itemID, "fc_call_") { typed["item_id"] = "fc_<dynamic>" }
        for _, child := range typed { normalizeCtoxGeminiStream(child) }
    case []any:
        for _, child := range typed { normalizeCtoxGeminiStream(child) }
    }
}
